    <h2>Evaporative Dissipation vs. Closed-Loop Dry Cooling</h2>
    <p>Traditional cooling towers transfer heat via the latent heat of vaporization. While thermally efficient, an average hyperscale campus running open-loop towers consumes between 1 and 5 million gallons of water every day. In water-stressed basins, this high demand can strain municipal supplies.</p>
    <p>In contrast, closed-loop Direct-to-Chip (DLC) architectures mount sealed micro-channel plates directly to processors. Hot dielectric or treated water circulates to external closed-circuit dry coolers, rejecting heat into ambient air with zero operational water loss (WUE = 0.0 L/kWh).</p>

  <script>
    const loadSlider = document.getElementById('it-load');
    const wueSlider = document.getElementById('evap-wue');
    const loadDisplay = document.getElementById('load-display');
    const wueDisplay = document.getElementById('wue-display');
    const evapLossDisplay = document.getElementById('annual-water-evap');
    const savedDisplay = document.getElementById('water-saved');

    function calculateImpact() {
      const mw = parseFloat(loadSlider.value);
      const wue = parseFloat(wueSlider.value);
      loadDisplay.innerText = `${mw} MW`;
      wueDisplay.innerText = `${wue.toFixed(1)} L/kWh`;

      // Calculation: MW * 1000 kW * 8760 hours/yr * WUE (L/kWh)
      const annualLiters = mw * 1000 * 8760 * wue;
      const millionLiters = (annualLiters / 1000000).toFixed(1);

      evapLossDisplay.innerText = `${millionLiters}M L`;
      savedDisplay.innerText = `${millionLiters}M L`;
    }

    loadSlider.addEventListener('input', calculateImpact);
    wueSlider.addEventListener('input', calculateImpact);
    calculateImpact();
  </script>
