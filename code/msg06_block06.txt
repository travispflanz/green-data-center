    // Theme Engine (Dark/Light Mode)
    const themeBtn = document.getElementById('theme-btn');
    const htmlEl = document.documentElement;
    const currentTheme = localStorage.getItem('theme') || (window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light');
    htmlEl.setAttribute('data-theme', currentTheme);

    themeBtn.addEventListener('click', () => {
      const nextTheme = htmlEl.getAttribute('data-theme') === 'dark' ? 'light' : 'dark';
      htmlEl.setAttribute('data-theme', nextTheme);
      localStorage.setItem('theme', nextTheme);
    });

    // Cloudflare D1 Native Subscription Handler
    const form = document.getElementById('newsletter-form');
    const msgEl = document.getElementById('form-msg');
    const btn = document.getElementById('subscribe-btn');

    form.addEventListener('submit', async (e) => {
      e.preventDefault();
      btn.disabled = true;
      btn.innerText = 'Submitting...';
      msgEl.style.display = 'none';

      try {
        const res = await fetch('/api/subscribe', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ email: document.getElementById('subscriber-email').value, source: 'index' })
        });
        const data = await res.json();
        msgEl.style.display = 'block';
        msgEl.style.color = data.success ? 'var(--accent-green)' : 'red';
        msgEl.innerText = data.message;
        if (data.success) form.reset();
      } catch (err) {
        msgEl.style.display = 'block';
        msgEl.style.color = 'red';
        msgEl.innerText = 'Network error. Please try again.';
      } finally {
        btn.disabled = false;
        btn.innerText = 'Subscribe';
      }
    });

    // Quick Search Client Filter
    document.getElementById('quick-search').addEventListener('input', (e) => {
      const query = e.target.value.toLowerCase();
      document.querySelectorAll('#searchable-cards article').forEach(card => {
        card.style.display = card.innerText.toLowerCase().includes(query) ? 'block' : 'none';
      });
    });
  </script>
